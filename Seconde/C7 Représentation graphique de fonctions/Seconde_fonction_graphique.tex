\section{Définir une fonction}
% Fonction à valeurs réelles définie sur un intervalle ou une réunion finie d’intervalles de ℝ.
% Recherche de domaine d’étude (ensemble de définition).
% Modéliser par des fonctions des situations issues des mathématiques, des autres disciplines ou de la vie courante ou citoyenne.
% Fonctions valeur absolue, carré, inverse : définitions et courbes représentatives.

\begin{defi}{Fonction réelle}{}
    On se donne un ensemble $D$ appelé \textbf{ensemble de définition} (ou ensemble de départ).

    Définir une \textbf{fonction} réelle sur $D$, c'est associer à tout élément $x\in D$ un unique réel que l'on note $f(x)$.

    Une fonction $f$ définie sur $D$ est généralement donnée:
    \begin{itemize}
        \item soit par une \textbf{expression algébrique}
        \item soit par un \textbf{tableau de valeurs}.
        \item soit par une \textbf{courbe représentative}
    \end{itemize}
\end{defi}

\begin{rem}{}{}
    On peut considérer qu'une fonction est une \og machine \fg. Chaque fois qu'on lui donne le même nombre $x$ en entrée, elle produit le même résultat $f(x)$.
\end{rem}

\begin{ex}{}{}
    Sur l'ensemble $D_f$ des mots de la langue française, on peut définir la fonction $f$  qui à chaque mot associe le nombre de lettres.

    Alors pour $x=\text{banane}$, on a $f(x) = 6$. On a aussi $f(\text{pomme}) = 5$ ou encore $f(\text{cerise})=6$.
\end{ex}

\begin{defi}{Fonction valeur absolue}{}
    On définit $g(x) = |x|$ sur l'ensemble de départ $D_g=\R$.
\end{defi}

\begin{ex}{}{}
    On peut calculer $g(2)=4$, $g(-5)=25$.

    $g$ est alors définie par l'expression algébrique $|x|$ en fonction d'un \textbf{paramètre} $x\in D_g$.
\end{ex}

\begin{ex}{}{}
    On définit la fonction $h$ sur l'ensemble $D_h = \fo{1}{6} \cap \Z$. On a $D_h = \{1;\quad 2;\quad 3;\quad 4;\quad 5\}$.

    \begin{tabular}{*{6}{|c}|}
        \hline
        $x\in\D_h$ & 1 & 2 & 3 & 4 & 5 \\
        \hline
        $f(x)$ & -1 & -2 & 6 & 8 & 10 \\
        \hline
    \end{tabular}

    $h$ est alors définie par le tableau de valeurs.
\end{ex}

\begin{defi}{Image, antécédent}{}
    \begin{itemize}
        \item
              Pour $x\in D_f$, on dit que $f(x)$ est l'\textbf{image} de $x$ par la fonction $f$
        \item
              Pour $y\in\R$. S'il existe $x\in D_f$ tel que $f(x)=y$, on dit que $x$ est \underline{un} \textbf{antécédent} de $y$ par la fonction $f$.
    \end{itemize}
\end{defi}

\begin{ex}{}{}
    Soit $g$ la fonction définie sur $D_g=\R$ par $g(x)=x^2$.
    \begin{itemize}
        \item  L'image de $5$ est $g(5)=5^2=25$.
        \item  Pour trouver les antécédents de $16$, on résout l'\textbf{équation} $g(x)=16$, d'inconnue $x\in D_g$. C'est-à-dire qu'on cherche les éléments $x\in D_g$ tels que $x^2=16$. On trouve alors deux antécédents, $4$ et $-4$. Car $g(-4)=(-4)^2=16$.
        \item  Il n'y a pas d'antécédent de $-1$ car l'équation $g(x)=-1$ n'a pas de solution. Un carré est toujours positif.
    \end{itemize}
\end{ex}

\begin{rems}{}{}
    \begin{itemize}
        \item
        Le mot \og antécédent \fg{} signifie \og qui se produit avant dans le temps. Ce nombre est antérieur au calcul par la fonction.
        \item
        Pour toute fonction $f$, un nombre $x$ a une et une seule image par $f$.

        Par contre, chaque nombre $y$ peut avoir un antécédent, plusieurs antécédents ou aucun antécédent.
    \end{itemize}

\end{rems}

\begin{notation}{}{}
    Une fonction (réelle) $f$ définie sur $D_f$ se note $f:D_f\rightarrow\R$.

    Si la fonction est donnée par une expression algébrique, alors on note $\fonction{f}{D_f}{\R}{x}{f(x)}$.

    On fera attention à la différence entre la flèche $\rightarrow$ pour l'ensemble de définition et la flèche $\mapsto$ pour l'expression algébrique.
\end{notation}

\begin{defi}{Fonction carrée}{}
    La fonction $g$ définie ci-dessus s'appelle la \textbf{fonction carrée} et se note $\fonction{g}{\R}{\R}{x}{x^2}$.
\end{defi}


\begin{methode}{Trouver l'ensemble de définition maximal à partir d'une expression algébrique}{}
    A partir d'une expression algébrique $f(x)$, on veut généralement trouver l'ensemble $D\subset\R$ maximal (le plus grand possible), tel que l'expression f(x) ait un sens pour tous les $x\in D$.

    Il faut faire attention aux abscisses $x\in\R$ qui meneraient à une des situations suivantes:
    \begin{itemize}
        \item Une division par zéro
        \item Une racine carrée d'un nombre négatif
    \end{itemize}

    Ces valeurs $x$ sont dites \textbf{valeurs interdites}.
\end{methode}

\begin{ex}{}{}
    La fonction $f:x \mapsto \dfrac{1}{2x-4}$ a pour ensemble de définition $] - \infty ; 2\; [\; \cup ]\;2 ;+ \infty [$ (tous les réels sauf 2).
    \begin{itemize}
        \item  En effet, l'expression $\dfrac{1}{2x-4}$ n'a de sens que pour les valeurs de $x$ telles que $2x-4\neq 0$ (car le dénominateur d'une fraction ne peut être égal à 0), c'est-à-dire pour $x \neq 2$,
        \item  On dira aussi que $2$ est une \textbf{valeur interdite} pour la fonction $f$.
    \end{itemize}
\end{ex}


\section{Représentation graphique}
% Courbe représentative : la courbe d’équation 𝑦 = ƒ (𝑥) est l’ensemble des points du plan dont les coordonnées (𝑥, 𝑦) vérifient 𝑦 = ƒ (𝑥).
% Exploiter l’équation 𝑦 = ƒ (𝑥) d’une courbe : appartenance, calcul de coordonnées.

\begin{defi}{Courbe}{}
  Soit $f:D\rightarrow\R$ une fonction. La \textbf{courbe représentative} ou le \textbf{graphe} de la fonction $f$ est l'ensemble des points du plan d'abscisse $x\in D$ et d'ordonnée $f(x)$. Ce sont donc les points de coordonnées $(x; f(x))$ pour $x\in D$.
\end{defi}

\begin{methode}{Tracer \textbf{l'allure} de la courbe d'une fonction}{}
    Soit $f$ une fonction définie sur un ensemble $D$ infini (par exemple $D=\ff{0}{1}$ ou $D=\R$).
    \begin{enumerate}
        \item
        On commence par créer un tableau de valeurs. Généralement, on prend des abscisses $x$ régulièrement espacées dans $D$.
        \item
        On place les points aux coordonnées $(x; f(x))$.
        \item
        A main levée, on essaie de passer par les points placés avec une courbe \og lisse\fg{}.
    \end{enumerate}
\end{methode}

\begin{ex}{Fonction carrée}{}
    \begin{tabular}{*{12}{|c}|}
    \hline
    $x\in\ff{-1}{1}$ & -1 & -0.8 & -0.6 & -0.4 & -0.2 & 0 & 0.2 & 0.4 & 0.6 & 0.8 & 1 \\
    \hline
    $f(x)$ & 1 & 0.64 & 0.36 & 0.16 & 0.04 & 0 & 0.04 & 0.16 & 0.36 & 0.64 & 1 \\
    \hline
    \end{tabular}

    La courbe représentative de la fonction carré est une \textbf{parabole} de \textbf{sommet} l'origine $O$ du repère.

\begin{prop}{}{}
  Soit $f$ une fonction, $\mathcal{C}_f$ sa courbe et $A(x,y)$ un point.

  $A\in\mathcal{C}_f$ si et seulement si $f(x)=y$
\end{prop}

\section{Tableau de signe}
% Tableau de signes pour une fonction produit ou quotient.

\begin{defi}{Tableau de signe}{}
  Soit $f:D\rightarrow\R$ une fonction. Dans un tableau de signe, pour chaque valeur $x\in D$, on indique si $f(x)$ est strictement négative, nulle ou strictement positive. La double barre correspond à une valeur interdite, le hachurage correspond à un intervalle interdit.
\end{defi}

\begin{ex}{Fonction valeur absolue}{}
    Prenons la fonction valeur absolue. On sait que pour $x\in\R$, $|x|\ge0$. De plus, l'équation $|x|=0$ admet une unique solution, $S=\{0\}$. D'où le tableau de signes suivant:

    \begin{tikzpicture}
   \tkzTabInit{$x$ / 1 , $|x|$ / 1}{$-\infty$, $0$, $+\infty$}
   \tkzTabLine{,+, z, +,}
    \end{tikzpicture}
\end{ex}

\begin{defi}{Fonction inverse}{}
  On définit la fonction inverse par \fonction{f}{\R^*}{\R}{x}{\frac{1}{x}}. Elle n'est pas définie en $0$ car on ne peut pas diviser par $0$.
\end{defi}

\begin{ex}{Fonction inverse}{}
    On résout $\frac{1}{x}>0$ et on trouve $x>0$.

    \begin{tikzpicture}
   \tkzTabInit{$x$ / 1 , $\frac{1}{x}$ / 1}{$-\infty$, $0$, $+\infty$}
   \tkzTabLine{,-, d, +,}
    \end{tikzpicture}
\end{ex}

\begin{methode}{Dresser le tableau de signes d'un produit ou d'un quotient}{}
    On s'interesse à une fonction $f$ définie comme un produit $f=g\times h$ ou comme un quotient $f=\frac{g}{h}$ de deux fonctions $g$ et $h$.

    \begin{enumerate}
        \item On dresse dans la première ligne, le tableau de signes de $g$.
        \item On dresse ensuite le tableau de signes de $h$.
        \item On déduit alors le signe du produit ou du quotient. Si le dénominateur $h$ s'annule alors il y a une valeur interdite.
    \end{enumerate}
\end{methode}

\begin{ex}{}{}
  Soit $f$ définie pour $x\in\R$ par $f(x) = \frac{2x+3}{x^2}$.

    \begin{tikzpicture}
    \tkzTabInit{$x$ / 1 , $2x+3$ / 1, $x^2$ / 1, $f(x)$ / 1}{$-\infty$, $-\frac{2}{3}$, $0$, $+\infty$}
    \tkzTabLine{,-, z, +, , +,}
    \tkzTabLine{,+, ,+, z, +,}
    \tkzTabLine{,-, z, +, d, +,}
    \end{tikzpicture}
\end{ex}

\section{Parité d'une fonction}
	\subsection{Fonction paire}

	\begin{defi}{Fonction paire}{}
	Une fonction $f$ définie sur un ensemble $I$ symétrique par rapport à $0$ est \bf{paire} si et seulement si pour tout $x \in I$ : $$f(-x)=f(x)$$
	\end{defi}

    \begin{ex}{}{}
    La fonction carrée est paire. Pour $x\in\R$, on a $f(-x)=(-x)^2=x^2=f(x)$.
    \end{ex}

	\begin{prop}{}{}
	Dans un repère orthogonal, la courbe représentative d'une fonction paire est symétrique par rapport à l'axe des ordonnées.
	\end{prop}

    \begin{dem}{}{}
        Soit $f$ une fonction paire. Soit un point $A(x,y)$ de la courbe représentative $\mathcal{C}_f$ avec $y=f(x)$. Le symétrique par rapport à l'axe des ordonnées est le point $B(-x,y)$. Il suffit de vérifier que B appartient bien à $\mathcal{C}_f$.

    Or $f(-x)=f(x)=y$. Donc $B\in\mathcal{C}_f$. La courbe est bien symétrique par rapport à l'axe des ordonnées.
    \end{dem}

	\subsection{Fonction impaire}

	\begin{defi}{Fonction impaire}{}
		Une fonction $f$ définie sur $I$, symétrique par rapport à $0$, est \bf{impaire} si et seulement si pour tout $x \in I$ : $$f(-x)=-f(x)$$
	\end{defi}

    \begin{ex}{}{}
      La fonction inverse est impaire. Pour $x\in\R^*$, on a $f(-x)=\frac{1}{-x}=-\frac{1}{x}=-f(x)$.
    \end{ex}

	\begin{prop}{}{}
	La courbe représentative d'une fonction impaire est symétrique par rapport à l'origine $O$ du repère. Il s'agit d'une symétrie centrale.
	\end{prop}

	\begin{methode}{Montrer qu'une fonction est paire ou impaire}{}
		\begin{itemize}
		\item On vérifie que l'ensemble de définition de la fonction est symétrique par rapport à 0. Si l'ensemble de définition n'est pas symétrique par rapport à 0, la fonction n'est ni paire ni impaire.

		\item Il peut s'avérer utile de tracer la courbe représentative de $f$ à la calculatrice.
		Si la courbe semble symétrique par rapport à l'axe des ordonnées, on pourra supposer que la fonction est paire.
		Si la courbe semble symétrique par rapport à l'origine, on pourra supposer que la fonction est impaire.

		\item On démontre que $f$ est paire ou impaire par le calcul:
		\begin{enumerate}
		\item On calcule $f\left(-x\right)$ en remplaçant $x$ par $\left(-x\right)$ dans l'expression de $f\left(x\right)$.
		\item Si on trouve que $f\left(-x\right)=f\left(x\right)$, alors $f$ est paire.
		\item Si on trouve que $f\left(-x\right)=-f\left(x\right)$, alors $f$ est impaire.
		\end{enumerate}
		\end{itemize}
		\end{methode}
