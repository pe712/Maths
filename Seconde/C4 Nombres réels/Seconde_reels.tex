\subsection{Intervalles}
% Intervalles de ℝ. Représentation graphique, notations du type [a, +∞[, ] –∞, a], [a, b], etc.
% Représenter un intervalle de la droite numérique. Déterminer si un nombre réel appartient à un intervalle donné.

\begin{defi}{Intervalle borné}{}
	Soit $a, b\in\R$ des réels avec $a\le b$. Soit $I$ l'ensemble des nombres réels compris entre $a$ et $b$. On dit que $I$ est un \bf{intervalle borné} dont les bornes sont $a$ et $b$. Les bornes peuvent appartenir à l'intervalle ou pas.
\end{defi}

\begin{notation}{}{}
	On note un intervalle avec les extrémités entre crochets. Le crochet est tourné vers l'interieur lorsque l'extrémité appartient à l'intervalle.

Il y a donc quatre types d'intervalles bornés:
	\begin{itemize}
		\item L'ensemble des nombres réels $x\in\R$ tels que $a\le x\le b$. On le note $I = \ff{a}{b}$. Les deux extrémités appartiennent à $I$.
		\item  L'ensemble des nombres réels $x\in\R$ tels que $a<x\le b$. On le note $I = \of{a}{b}$. $b$ appartient à $I$ mais pas $a$.
		\item L'ensemble $I = \fo{a}{b}$.
		\item L'ensemble $I = \oo{a}{b}$.
	\end{itemize}
\end{notation}

\begin{ex}{}{}
    	$I = \ff{0}{5}$ est l'ensemble des nombres entre 0 et 5. Il y a par exemple $0, \quad 3/5, \quad 4.1235456 \quad$ et $\sqrt{2}$ dedans.


	$J =\fo{0}{\pi}$ est l'ensemble des nombres positifs strictement plus petits que $\pi$.

\begin{rem}{}{}
	Un intervalle est donc une partie "sans trous" de l'ensemble des nombres réels $\R$.
\end{rem}




\begin{defi}{Intervalle non borné}{}
	Soit $a\in\R$ un réel. Soit $I$ l'ensemble des nombres réels plus grand que $a$. La borne $a$ peut appartenir à $I$ ou pas. De même, on définit l'ensemble $J$ des nombres réels plus petis que $a$.

	On dit que $I$, $J$ et l'ensemble $\R$ entier sont des \bf{intervalles non bornés}.
\end{defi}

\begin{notation}{}{}
	Pour un intervalle non borné, on remplace une extrémité par $+\infty$ ou $-\infty$. Le crochet est toujours ouvert de ce côté, car l'infini n'est pas un nombre réel.
\end{notation}

\begin{exs}{}{}

\begin{prop}{}{}
	Un intervalle borné correspond à un segment de la droite réelle.

	Un intervalle non borné correspond à une demi-droite ou a la droite réelle entière.
\end{prop}

\section{Valeur absolue}
% Notation en valeur absolue |a| pour la distance de a à 0. Distance entre deux nombres réels.
% Relation √a2 = |a|.

\begin{defi}{Valeur absolue}{}
Soit $a$ un nombre réel.

 On appelle valeur absolue de $a$, la distance entre $a$ et $0$. On note cette distance $|a|$. C'est donc un nombre positif.
\end{defi}

\smallskip
\begin{ex}{}{}
Essayons de déterminer $|-12|$.\medskip

 On sait d'après la définition précédente que $|-12|$ est la \\distance entre $0$ et $-12$. On représente la droite des réels, \\puis (en rouge) la distance entre $0$ et $-12$:

\begin{prop}{}{}
  Soit $a\in\R$.
  \begin{itemize}
    \item Si $a\ge0$, alors $|a| =  a$.
    \item si $a\le0$, alors $|a| =  -a$
  \end{itemize}
\end{prop}

\bigskip
 Avec la notation valeur absolue, il est possible de calculer la distance entre deux nombres réels:

\begin{prop}{}{}
Soient $a$ et $b$ deux nombres réels. $|a-b|$ ou $|b-a|$ est la distance entre $a$ et $b$
\end{prop}

\begin{rem}{}{}
  Dans le cas ou $b=0$, on retrouve la définition de la valeur absolue: la distance de $a$ à 0 est $|a-0|$ c'est à dire $|a|$.
\end{rem}

\begin{prop}{Inéquation $|x-a|\le r$}{}
	Soit $a\in\R$, $r\in\R$. On considère l'inéquation $|x-a|\le r$ dont l'inconnue est le réel $x\in\R$.

	\begin{itemize}
		\item Si $r\ge0$, l'ensemble des solutions est l'intervalle $S=\ff{a-r}{a+r}$.
		\item Si $r<0$, il n'y a pas de solutions. On note $S=\emptyset$.
	\end{itemize}

		\draw (-2,-1) grid (7,1);
    	\draw[->, thick] (-2,0) -- (7,0);
    	\draw[thick] (0, 0.1) -- (0, -0.1) node [above left] {0};
    	\draw[thick] (1, 0.1) -- (1, -0.1) node [above left] {1};
    	\draw[red, very thick] (-1, 0) --  node [below] {$\ff{a-r}{a+r}$} (5, 0);
    	\draw[red, very thick] (2, 0.1) --  node [above] {$a=2$} (2, -0.1);
    	\draw[red, very thick] (-0.8, 0.5) --  (-1, 0.5) -- (-1, -0.5) -- (-0.8, -0.5);
    	\draw[red, very thick] (4.8, 0.5) --  (5, 0.5) -- (5, -0.5) -- (4.8, -0.5);
		\draw[blue, very thick, <->] (2, -0.7) -- node[below]{$r=3$} (5, -0.7);
	\end{tikzpicture}
\end{prop}

\begin{dem}{}{}
	Soit $x\in\R$.

	Le nombre $x$ est solution si sa distance à $a$ est inférieure à $r$.

	\begin{itemize}
		\item Si $r\ge0$, il s'agit donc de l'ensemble $\ff{a-r}{a+r}$.

		Les cas extrêmes sont les cas où $|x-a|=r$. Ces cas correspondent aux deux bornes $a-r$ et $a+r$ de notre intervalle.

		\item Si $r<0$, c'est impossible, car une distance est toujours positive.
	\end{itemize}
\end{dem}

\begin{defi}{Intervalle centré}{}
	Un intervalle de la forme $\ff{a-r}{a+r}$ est appelé un intervalle centré en $a$ et d'amplitude (ou de précision) $r$.
\end{defi}

\begin{ex}{}{}
	Un encadrement de $\sqrt{7}\approx 2.64575131$ à $10^-4$ près est donné par $ 2.6457 \le \sqrt{7} \le 2.6458$.

	On peut écrire cela sous forme d'intervalle centré. Soit $I=\ff{2.64575-0.00005}{2.64575+0.00005}=\ff{a-r}{a+r}$ avec $a=2.64575$ et $r=0.00005$. On a $\sqrt{7} \in I$.
\end{ex}

\section{Puissances d'un nombre}
% Règles de calcul sur les puissances entières relatives, sur les racines carrées.

\begin{defi}{Puissance d'un nombre}{}
  Soient $n\in\N^*$ et $x\in\R$.

  \begin{itemize}
	\item
	$x^n= x\times x \times \dots\times x$
	\item Pour $x\ne0$, $x^{-n}=\frac{1}{x^n}$
  \end{itemize}
\end{defi}

\begin{rems}{}{}
	\begin{itemize}
		\item
		Par convention, pour $x\ne0$, $x^0=1$.
		\item
		Dans le cas particulier $n=-1$ on obtient $x^{-1}=\frac{1}{x}$.
	\end{itemize}
\end{rems}

\textit{\og Gros, c'est la puissance, rien que la puissance, respecte le protocole, gros y'a pas d'secret.\fg{}} --- La puissance, MHD

\begin{prop}{Règles de calcul avec les puissances}{}
	Soient $a\in\R$, $b\in\R$, $n\in\N$ et $m\in\N$. On a :
	\begin{itemize}
		\item le produit et quotient d'un même nombre: $a^n\times a^m = a^{n+m}$ et $\frac{a^n}{a^m} = a^{n-m}$.
		\item le produit et le quotient à un même exposant: $(a\times b)^n = a^n \times b^n$ et $(\frac{a}{b})^n=\frac{a^n}{b^n}$.
		\item la puissance de puissance: $(a^n)^m = a^{nm}$
	\end{itemize}
\end{prop}

\begin{dem}{}{}
	Prouvons que $a^n\times a^m = a^{n+m}$. On part du membre de gauche et on calcule:

	$a^n\times a^m = (\underbrace{a\times a\times \dots\times a}_{n \text{ fois}}) \times (\underbrace{a\times a\times \dots\times a}_{m \text{ fois}}) = \underbrace{a\times a\times \dots\times a}_{n+m \text{ fois}} = a^{n+m}$

	Prouvons que $(a\times b)^n = a^n \times b^n$. On procède de même:

	$(a\times b)^n = \underbrace{(a\times b)\times(a\times b)\times \dots \times(a\times b)}_{n \text{ fois}} =  (\underbrace{a\times a\times \dots\times a}_{n \text{ fois}}) \times (\underbrace{b\times b\times \dots\times b}_{n \text{ fois}}) = a^n \times b^n$

	Prouvons que $(a^n)^m = a^{nm}$. On procède de même:

	$(a^n)^m = \underbrace{(a^n)\times(a^n)\times(a^n)\times\dots\times(a^n)}_{m \text{ fois}}  = a^{\underbrace{n+n+\dots+n}_{m \text{ fois}}}= a^{n\times m}$


\end{dem}

\begin{ex}{}{}
	$A = 2^5\times2^7\times\frac{1}{8^2}\times 2^{-3} = \frac{2^5\times2^7\times2^{-3}}{(2^3)^2}  = \frac{2^{5+7-3}}{2^6} = \frac{2^9}{2^6} = 2^{9-6} = 2^3 = 8$
\end{ex}

\section{Racine carrée d'un nombre}
\begin{defi}{Racine carrée}{}
	Soit $x\in\R$ avec $x\ge 0$. La racine carrée de $x$ est l'unique nombre positif qui mis au carré donne $x$. On le note $\sqrt{x}$.
\end{defi}

\begin{ex}{}{}
On a $\sqrt{64}=8$ car $8^2=64$.

On peut vérifier que $\sqrt{2.25} = 1.5$. En effet, $1.5^2 = 2.25$. Donc 1.5 est bien l'unique nombre positif qui mis au carré vaut 2.25. D'où $\sqrt{2.25} = 1.5$.
\end{ex}

\begin{rem}{}{}
	Même si $x\in\N$, $\sqrt{x}$ peut être irrationnel. C'est le cas de $\sqrt{2}$.
\end{rem}

\begin{ex}{}{}
  Calculons $\sqrt{a^2}$ pour différentes valeurs de $a$.

  Pour $a=4$, on trouve $\sqrt{4^2} = \sqrt{16} = 4$.

  Pour $a=-3$, on trouve $\sqrt{(-3)^2}=\sqrt{9}=3$.
\end{ex}

\begin{prop}{}{}
    Soit $a\in\R$. Alors $|a|=\sqrt{a^2}$.
\end{prop}

\begin{dem}{}{}
    Soit $a\in\R$. On distingue deux cas.
  \begin{itemize}
    \item Si $a\ge0$, alors d'une part $\sqrt{a^2} = a$ car $a$ est positif. C'est donc bien l'unique nombre positif qui mis au carré donne $a^2$. Dautre part, $|a| =  a$. Donc on a bien $\sqrt{a^2}=|a|$.
    \item Si $a\le0$, on pose $b=-a$. $b$ est donc positif. On calcule alors d'une part, $\sqrt{a^2} = \sqrt{(-a)^2}=\sqrt{b^2} = b$ car $b$ est positif. Dautre part, $|a| =  -a = b$. Donc on a bien $\sqrt{a^2}=|a|$.
  \end{itemize}

\end{dem}
